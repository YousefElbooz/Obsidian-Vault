```dataviewjs
const gcal = this.app.plugins.plugins["google-calendar"]?.api;

if (!gcal) {
    dv.paragraph("⚠️ **Google Calendar plugin not found.** Please ensure it is enabled.");
} else {
    // 1. Grab the date from the current file context
    const fileDateText = dv.current().file.day ? dv.current().file.day.toISODate() : dv.current().file.name;
    const referenceDate = moment(fileDateText);

    // 2. Fetch events
    const events = await gcal.getEvents({
        startDate: referenceDate.clone().startOf('week'),
        endDate: referenceDate.clone().endOf('week')
    });

    const sleepDataMap = {};
    const dailySegments = {}; // Maps dates to their specific sleep blocks

    events.forEach(e => {
        if (e.summary && e.summary.toLowerCase().includes("sleep")) {
            let start = moment(e.start.dateTime || e.start.date);
            let end = moment(e.end.dateTime || e.end.date);
            let totalDuration = moment.duration(end.diff(start)).asHours();
            
            // For the daily total calculation, we assign the whole duration to the start date 
            // (or you can split it, but keeping it on start date is standard for sleep tracking)
            let dateKey = start.format("YYYY-MM-DD");
            sleepDataMap[dateKey] = (sleepDataMap[dateKey] || 0) + totalDuration;

            // --- Split events for the multi-row timeline ---
            let startDateStr = start.format("YYYY-MM-DD");
            let endDateStr = end.format("YYYY-MM-DD");
            let startHourFloat = start.hours() + start.minutes() / 60;
            
            if (startDateStr === endDateStr) {
                // Event starts and ends on the same day
                if (!dailySegments[startDateStr]) dailySegments[startDateStr] = [];
                dailySegments[startDateStr].push({ start: startHourFloat, duration: totalDuration });
            } else {
                // Event crosses midnight, split into two blocks
                let durationBeforeMidnight = 24 - startHourFloat;
                if (!dailySegments[startDateStr]) dailySegments[startDateStr] = [];
                dailySegments[startDateStr].push({ start: startHourFloat, duration: durationBeforeMidnight });

                let endHourFloat = end.hours() + end.minutes() / 60;
                if (!dailySegments[endDateStr]) dailySegments[endDateStr] = [];
                dailySegments[endDateStr].push({ start: 0, duration: endHourFloat });
            }
        }
    });

    const sortedDates = Object.keys(sleepDataMap).sort().reverse();

    if (sortedDates.length === 0) {
        dv.paragraph("🌙 *No sleep events found for the week of this file.*");
    } else {
        let totalHours = 0;
        let minGoal = 6;      
        let dangerLow = 4;
        let dangerHigh = 10;
        let progressMax = 12; 
        let tableData = [];
        let oversleepDays = 0;

        sortedDates.forEach(date => {
            let duration = sleepDataMap[date];
            totalHours += duration;
            if (duration > dangerHigh) oversleepDays++;

            let percent = Math.min((duration / progressMax) * 100, 100);
            let barColor = "#4caf50"; 
            
            if (duration < dangerLow) barColor = "#f44336"; 
            else if (duration > dangerHigh) barColor = "#ffeb3b"; 
            else if (duration < minGoal) barColor = "#ff9800"; 

            let progressBar = `<div style="width:100px; background:rgba(255,255,255,0.1); border-radius:5px; height:8px; display:inline-block; vertical-align:middle;"><div style="width:${percent}%; background:${barColor}; height:100%; border-radius:5px;"></div></div>`;

            let diff = duration - minGoal;
            let diffText = diff >= 0 ? `<span style="color:#4caf50">+${diff.toFixed(1)}</span>` : `<span style="color:#f44336">${diff.toFixed(1)}</span>`;

            tableData.push([
                moment(date).format("ddd, MMM DD"), 
                `**${duration.toFixed(2)}h**`, 
                diffText,
                progressBar 
            ]);
        });

        const avg = parseFloat((totalHours / sortedDates.length).toFixed(2));
        
        let sumOfSquaredDifferences = 0;
        sortedDates.forEach(date => {
            let diff = sleepDataMap[date] - avg;
            sumOfSquaredDifferences += (diff * diff);
        });
        const stdDev = Math.sqrt(sumOfSquaredDifferences / sortedDates.length);
        
        let qualityStatus = ""; let qualityColor = ""; let qualityMessage = "";
        if (stdDev <= 0.8) {

            qualityMessage = "> [!success] **Routine Quality:** Excellent. Your sleep hours are highly consistent, which is great for maintaining deep focus during development and study sessions.";
        } else if (stdDev <= 1.5) { 
            qualityStatus = "Fair"; qualityColor = "#ffeb3b"; 
            qualityMessage = "> [!warning] **Routine Quality:** Fair. Your sleep durations fluctuate a bit. Tightening up your bedtime could yield better energy levels.";
        } else { 
            qualityStatus = "Poor"; qualityColor = "#f44336"; 
            qualityMessage = "> [!danger] **Routine Quality:** Inconsistent. Your sleep hours vary wildly from day to day. Try locking in a stricter schedule to avoid burning out your cognitive load.";
        }

        let cardContainer = dv.el("div", "", { attr: { style: "display: flex; gap: 10px; margin-bottom: 20px; flex-wrap: wrap;" }});
        const createCard = (label, value, color) => {
            let card = cardContainer.createEl("div", { attr: { style: `flex: 1; min-width: 100px; background: rgba(255,255,255,0.05); padding: 12px; border-radius: 8px; border-left: 4px solid ${color};` }});
            card.createEl("div", { text: label, attr: { style: "font-size: 0.85em; color: gray; text-transform: uppercase; letter-spacing: 0.5px;" }});
            card.createEl("div", { text: value, attr: { style: "font-size: 1.4em; font-weight: bold; margin-top: 5px;" }});
        }

        createCard("Avg Sleep", `${avg}h`, avg < minGoal ? "#f44336" : "#4caf50");
        createCard("Consistency", qualityStatus, qualityColor);
        createCard("Total Hours", totalHours.toFixed(1), "#9c27b0");

		dv.paragraph(`#### 🧠 Weekly Analysis`);
        
        if (avg > dangerHigh) dv.paragraph(`> [!warning] **Duration Trend:** You are consistently oversleeping (averaging >${dangerHigh}h). This might lead to sleep inertia. Try capping sessions closer to 8 hours.`);
        else if (avg < dangerLow) dv.paragraph(`> [!danger] **Duration Trend:** You are averaging below ${dangerLow} hours. This severe lack of sleep will drastically impact your performance and focus.`);
        else if (avg < minGoal) dv.paragraph(`> [!warning] **Duration Trend:** You are below your ${minGoal}-hour baseline. Prioritize getting slightly more rest to optimize cognitive function.`);

                    <div style="width: 55px; font-size: 0.8em; color: var(--text-muted); text-align: right; padding-right: 10px; white-space: nowrap;">
```

```dataviewjs 
// 1. Setup & Target Date
const taskNotesFolder = '"TaskNotes"';
const scheduledBlocks = dv.current().timeblocks || [];

// Safely parse dates from Dataview (Luxon) or standard strings into Moment.js
function parseDate(dvDate) {
    if (!dvDate) return null;
    if (dvDate.toJSDate) return moment(dvDate.toJSDate()); // Handles Dataview Luxon objects
    return moment(new Date(dvDate)); // Handles raw strings
}

// Get the target date based on the current file's name (Assumes YYYY-MM-DD format)
const fileName = dv.current().file.name;
let targetMoment = moment(fileName, "YYYY-MM-DD", true);
if (!targetMoment.isValid()) {
    targetMoment = moment(); // Fallback to today if the file isn't named as a date
}
const targetDateStr = targetMoment.format("YYYY-MM-DD");

// Check if we are viewing "Today's" note for real-time tracking features
const realNow = moment();
const isTodayNote = targetMoment.isSame(realNow, 'day');
const currentMin = realNow.hours() * 60 + realNow.minutes();

let rawEntries = [];

// 2. Fetch pages and extract entries specifically for the target date
const pagesWithTime = dv.pages(taskNotesFolder).where(p => p.timeEntries);

pagesWithTime.forEach(page => {
    let entries = page.timeEntries;
    if (!Array.isArray(entries)) entries = [entries];
    
    entries.forEach(entry => {
        if (!entry.startTime) return;

        const startMoment = parseDate(entry.startTime);
        
        // Filter: ONLY grab entries that match the date of this specific note
        if (startMoment && startMoment.format("YYYY-MM-DD") === targetDateStr) {
            let isRunning = false;
            let endMoment = parseDate(entry.endTime);
            
            // If no end time, it's running. If looking at a past day, cap it.
            if (!endMoment || !endMoment.isValid()) {
                isRunning = isTodayNote; 
                endMoment = isTodayNote ? realNow : startMoment.clone().add(1, 'hours'); // Default 1h for abandoned past timers
            }

            let startMin = startMoment.hours() * 60 + startMoment.minutes();
            let endMin = endMoment.hours() * 60 + endMoment.minutes();

            // Handle midnight crossover
            if (endMin < startMin && !isRunning) endMin += 24 * 60;

            rawEntries.push({
                title: page.file.name.replace("To-Do ", ""),
                desc: entry.description || "",
                startTime: startMoment.format("HH:mm"),
                endTime: isRunning ? "Running" : endMoment.format("HH:mm"),
                startMin: startMin,
                endMin: endMin,
                isRunning: isRunning,
                color: "#4ade80" 
            });
        }
    });
});

// 3. Sort Ascending AND Merge consecutive entries for the same task
const mergedEntries = [];
if (rawEntries.length > 0) {
    // Sort chronologically (Ascending sequence)
    const sortedRaw = [...rawEntries].sort((a, b) => a.startMin - b.startMin);
    
    sortedRaw.forEach(entry => {
        const last = mergedEntries[mergedEntries.length - 1];
        if (last && last.title === entry.title && last.endMin >= entry.startMin) {
            last.endMin = Math.max(last.endMin, entry.endMin);
            last.endTime = entry.isRunning ? "Running" : parseDate(entry.endTime).format("HH:mm");
            last.isRunning = last.isRunning || entry.isRunning;
        } else {
            mergedEntries.push({ ...entry, isRealTime: true });
        }
    });
}

// 4. Combine Plan + Actual and Sort Again
const formattedBlocks = scheduledBlocks.map(b => {
    const [h, m] = b.startTime.split(':').map(Number);
    const [eh, em] = b.endTime.split(':').map(Number);
    return { ...b, isPlan: true, startMin: h * 60 + m, endMin: eh * 60 + em };
});

const allItems = [...formattedBlocks, ...mergedEntries].sort((a, b) => a.startMin - b.startMin);

// 5. Render UI
let html = `<div style="display: flex; flex-direction: column; gap: 12px; font-family: var(--font-interface);">`;

if (allItems.length === 0) {
    html += `<div style="text-align: center; color: var(--text-muted); padding: 20px;">No scheduled blocks or tracked time found for ${targetDateStr}.</div>`;
}

allItems.forEach(item => {
    // Logic for progress bars based on whether we are looking at today's note
    const isPast = !isTodayNote || (currentMin >= item.endMin && !item.isRunning);
    const isCurrent = isTodayNote && (item.isRunning || (currentMin >= item.startMin && currentMin < item.endMin));
    
    const blockColor = item.color || (item.isPlan ? "var(--interactive-accent)" : "#4ade80");
    const opacity = isPast ? "0.55" : "1";
    
    let progress = 0;
    if (isPast) progress = 100;
    else if (isCurrent) progress = ((currentMin - item.startMin) / (item.endMin - item.startMin)) * 100;

    const pulseStyle = item.isRunning ? `animation: pulse 2s infinite;` : "";
    const styleTag = item.isRunning ? `<style>@keyframes pulse { 0% { opacity: 0.8; } 50% { opacity: 1; filter: brightness(1.2); } 100% { opacity: 0.8; } }</style>` : "";

    html += `
    ${styleTag}
    <div style="background: rgba(255,255,255,0.03); border-radius: 8px; border-left: 6px ${item.isPlan ? 'solid' : 'dashed'} ${blockColor}; padding: 12px 16px; opacity: ${opacity}; ${pulseStyle}">
        <div style="display: flex; justify-content: space-between; align-items: flex-start; margin-bottom: 8px;">
            <div>
                <span style="font-size: 0.7em; font-weight: bold; color: ${blockColor}; text-transform: uppercase; letter-spacing: 0.5px;">
                    <span style="background: rgba(255,255,255,0.1); padding: 2px 4px; border-radius: 4px; margin-right: 4px;">✔</span>
                    ${item.isPlan ? "PLAN" : (item.isRunning ? "RECORDING" : "LOGGED")}
                </span>
                <div style="font-weight: 600; font-size: 1.1em; color: var(--text-normal); margin-top: 4px;">
                    ${item.title} ${isCurrent ? "⚡" : ""}
                </div>
                ${item.desc && !item.isPlan ? `<div style="font-size: 0.85em; color: var(--text-muted); margin-top: 2px;">↳ ${item.desc}</div>` : ""}
            </div>
            <div style="font-family: monospace; font-size: 0.95em; color: var(--text-muted); text-align: right;">
                ${item.startTime} - ${item.endTime}
            </div>
        </div>
        <div style="height: 5px; width: 100%; background: rgba(255,255,255,0.1); border-radius: 10px; overflow: hidden;">
            <div style="height: 100%; width: ${progress}%; background: ${blockColor}; transition: width 0.4s ease;"></div>
        </div>
    </div>`;
});

html += `</div>`;
dv.el("div", html);
```
